cmake_minimum_required(VERSION 3.20)
project(SkecsGaVulkan LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)

set(SKECS_GA_SLANG_ROOT "" CACHE PATH "Optional Slang 2025.6.1 SDK root for the Vulkan compiler adapter")
set(SKECS_GA_STARTUP_SPIRV "" CACHE FILEPATH "Prebuilt startup SPIR-V validated for Vulkan 1.1")
set(SKECS_GA_STARTUP_SPIRV_SHA256 "" CACHE STRING "SHA-256 of the prebuilt startup SPIR-V")

find_program(SKECS_GA_SPIRV_VAL spirv-val
    HINTS "$ENV{VULKAN_SDK}/bin" "$ENV{VULKAN_SDK}/Bin")

if(SKECS_GA_STARTUP_SPIRV)
    get_filename_component(startup_spirv "${SKECS_GA_STARTUP_SPIRV}" ABSOLUTE)
    if(NOT EXISTS "${startup_spirv}")
        message(FATAL_ERROR "Prebuilt startup SPIR-V does not exist: ${startup_spirv}")
    endif()
    string(TOLOWER "${SKECS_GA_STARTUP_SPIRV_SHA256}" startup_sha)
    string(LENGTH "${startup_sha}" startup_sha_length)
    if(NOT startup_sha MATCHES "^[0-9a-f]+$" OR NOT startup_sha_length EQUAL 64)
        message(FATAL_ERROR "Provide SKECS_GA_STARTUP_SPIRV_SHA256 from the validated prebuilt startup artifact")
    endif()
    file(SHA256 "${startup_spirv}" actual_sha)
    if(NOT actual_sha STREQUAL startup_sha)
        message(FATAL_ERROR "Prebuilt startup SPIR-V SHA-256 mismatch")
    endif()
else()
    find_program(SKECS_GA_SLANGC slangc REQUIRED
        HINTS "${SKECS_GA_SLANG_ROOT}/bin" "${SKECS_GA_SLANG_ROOT}/Bin" "$ENV{VULKAN_SDK}/bin" "$ENV{VULKAN_SDK}/Bin")
    if(NOT SKECS_GA_SPIRV_VAL)
        message(FATAL_ERROR "spirv-val is required to validate startup SPIR-V generated by slangc")
    endif()
    execute_process(COMMAND "${SKECS_GA_SLANGC}" -version
        RESULT_VARIABLE slang_result OUTPUT_VARIABLE slang_version ERROR_VARIABLE slang_version
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_STRIP_TRAILING_WHITESPACE)
    if(NOT slang_result STREQUAL "0" OR NOT slang_version STREQUAL "2025.6.1")
        message(FATAL_ERROR "Slang 2025.6.1 is required; found '${slang_version}'")
    endif()
    set(startup_spirv "${CMAKE_CURRENT_BINARY_DIR}/startup.spv")
    add_custom_command(OUTPUT "${startup_spirv}"
        COMMAND "${SKECS_GA_SLANGC}" "${CMAKE_CURRENT_SOURCE_DIR}/StartupProbe.slang"
            -entry main -target spirv -profile spirv_1_3 -o "${startup_spirv}"
        DEPENDS StartupProbe.slang "${SKECS_GA_SLANGC}"
        VERBATIM)
endif()

set(startup_header "${CMAKE_CURRENT_BINARY_DIR}/StartupSpirv.h")
add_custom_command(OUTPUT "${startup_header}"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT=${startup_spirv}" "-DOUTPUT=${startup_header}"
        "-DEXPECTED_SHA256=${startup_sha}" "-DSPIRV_VALIDATOR=${SKECS_GA_SPIRV_VAL}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/EmbedStartupSpirv.cmake"
    DEPENDS "${startup_spirv}" EmbedStartupSpirv.cmake
    VERBATIM)

add_library(ga-vulkan-native SHARED GaVulkanNative.cpp VulkanDispatch.cpp "${startup_header}")
target_compile_features(ga-vulkan-native PRIVATE cxx_std_20)
target_compile_definitions(ga-vulkan-native PRIVATE SKECS_GA_VK_BUILD)
target_include_directories(ga-vulkan-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ga-vulkan-native PRIVATE Vulkan::Vulkan Threads::Threads)
set_target_properties(ga-vulkan-native PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

if(ANDROID)
    set(SKECS_GA_STARTUP_GLSL_ES "" CACHE FILEPATH "Verified startup shader translated to GLSL ES 3.10")
    set(SKECS_GA_STARTUP_GLSL_ES_SHA256 "" CACHE STRING "SHA-256 of the prebuilt startup GLSL ES shader")
    if(SKECS_GA_STARTUP_GLSL_ES)
        get_filename_component(startup_gles "${SKECS_GA_STARTUP_GLSL_ES}" ABSOLUTE)
        set(startup_gles_header "${CMAKE_CURRENT_BINARY_DIR}/StartupGlslEs.h")
        add_custom_command(OUTPUT "${startup_gles_header}"
            COMMAND "${CMAKE_COMMAND}" "-DINPUT=${startup_gles}" "-DOUTPUT=${startup_gles_header}"
                "-DEXPECTED_SHA256=${SKECS_GA_STARTUP_GLSL_ES_SHA256}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/EmbedStartupGlslEs.cmake"
            DEPENDS "${startup_gles}" EmbedStartupGlslEs.cmake
            VERBATIM)
        add_library(ga-opengl-native SHARED GaOpenGlNative.cpp "${startup_gles_header}")
        target_compile_features(ga-opengl-native PRIVATE cxx_std_20)
        target_include_directories(ga-opengl-native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_link_libraries(ga-opengl-native PRIVATE EGL GLESv3 Threads::Threads)
        target_link_options(ga-opengl-native PRIVATE "LINKER:-z,max-page-size=16384" "LINKER:--exclude-libs,ALL")
        set_target_properties(ga-opengl-native PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    endif()
endif()

if(SKECS_GA_SLANG_ROOT)
    find_path(skecs_slang_include NAMES slang/slang.h slang.h
        PATHS "${SKECS_GA_SLANG_ROOT}" PATH_SUFFIXES include Include NO_DEFAULT_PATH REQUIRED)
    find_library(skecs_slang_library NAMES slang slang-compiler
        PATHS "${SKECS_GA_SLANG_ROOT}" PATH_SUFFIXES lib lib64 Lib NO_DEFAULT_PATH REQUIRED)
    add_executable(ga-slang-probe Probe.cpp VulkanDispatch.cpp)
    target_compile_features(ga-slang-probe PRIVATE cxx_std_20)
    target_compile_definitions(ga-slang-probe PRIVATE SKECS_GA_PROBE_VULKAN_ONLY)
    target_include_directories(ga-slang-probe PRIVATE "${skecs_slang_include}")
    target_link_libraries(ga-slang-probe PRIVATE Vulkan::Vulkan Threads::Threads "${skecs_slang_library}")
endif()

include(CTest)
if(BUILD_TESTING)
    option(SKECS_GA_SOFTWARE_SMOKE "Also register an explicit Vulkan software-device smoke test" OFF)
    add_executable(ga-native-probe NativeProbe.cpp)
    target_compile_features(ga-native-probe PRIVATE cxx_std_20)
    target_link_libraries(ga-native-probe PRIVATE ga-vulkan-native)
    add_test(NAME ga-vulkan-physical COMMAND ga-native-probe "${startup_spirv}")
    if(SKECS_GA_SOFTWARE_SMOKE)
        add_test(NAME ga-vulkan-software COMMAND ga-native-probe "${startup_spirv}" --software)
    endif()
endif()
